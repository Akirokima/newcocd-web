name: Publicar Bitácora desde Notion

# SOLO manual. No hay disparo programado: la Bitácora se publica cuando
# Andrés lo decide, nunca sola.
#
#   GitHub -> pestaña Actions -> "Publicar Bitácora desde Notion" -> Run workflow
#
# Lee Notion, escribe el markdown, regenera ideas.html y hace push. Marcar las
# entradas como "Publicada" en Notion va DESPUÉS del push, nunca antes.
on:
  workflow_dispatch:
    inputs:
      estados:
        description: 'Estados de Notion que se publican (separados por comas)'
        required: false
        default: 'Lista para publicar,Publicada'
      simulacion:
        description: 'Simular sin escribir nada (para ver qué entraría)'
        type: boolean
        required: false
        default: false

permissions:
  contents: write

concurrency:
  group: publicar-bitacora
  cancel-in-progress: false

jobs:
  publicar:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Leer entradas de Notion
        env:
          NOTION_TOKEN: ${{ secrets.NOTION_TOKEN }}
          NOTION_ESTADOS: ${{ inputs.estados }}
        run: |
          if [ "${{ inputs.simulacion }}" = "true" ]; then
            node scripts/sync-notion.mjs --dry-run
          else
            node scripts/sync-notion.mjs
          fi

      # Sin este paso el markdown estaría al día y la web seguiría igual.
      - name: Regenerar ideas.html
        if: ${{ inputs.simulacion != true }}
        run: node scripts/build-bitacora.mjs

      - name: Confirmar cambios
        id: push
        if: ${{ inputs.simulacion != true }}
        run: |
          if [ -z "$(git status --porcelain src/content/bitacora ideas.html)" ]; then
            echo "Sin cambios: la Bitácora ya estaba al día."
            echo "publicado=no" >> "$GITHUB_OUTPUT"
            exit 0
          fi
          git config user.name  "notion-sync"
          git config user.email "info@newcocd.com"
          git add src/content/bitacora ideas.html
          git commit -m "Bitácora: publicación desde Notion"
          git push
          echo "publicado=si" >> "$GITHUB_OUTPUT"
          echo "Publicado. Cambios:"
          git show --stat --oneline HEAD

      # Solo si el push ha terminado bien: si se marcara antes y el push
      # fallara, Notion diría "Publicada" y la web no tendría la entrada.
      - name: Marcar como Publicada en Notion
        if: ${{ inputs.simulacion != true && steps.push.outputs.publicado == 'si' }}
        env:
          NOTION_TOKEN: ${{ secrets.NOTION_TOKEN }}
        run: node scripts/marcar-publicadas.mjs
